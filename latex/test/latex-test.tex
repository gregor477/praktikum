\section*{1. Osnovne naloge}

\[
a_n = 3n - 2
\]
\vspace{2cm}

\item Dano je zaporedje: 2, 5, 8, 11, \dots

\begin{enumerate}

\end{enumerate}

\item Izračunaj:
\[
a_7 \quad \text{če je} \quad a_n = 4n + 1
\]
\vspace{2cm}

\end{enumerate}
